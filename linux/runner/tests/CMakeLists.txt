cmake_minimum_required(VERSION 3.13)
project(nauvim_renderer_tests LANGUAGES CXX)

enable_testing()
find_package(PkgConfig REQUIRED)
pkg_check_modules(GLIB REQUIRED IMPORTED_TARGET glib-2.0)
pkg_check_modules(GTK REQUIRED IMPORTED_TARGET gtk+-3.0)

add_executable(renderer_policy_test renderer_policy_test.cc ../renderer_policy.cc)
target_compile_features(renderer_policy_test PRIVATE cxx_std_14)
target_compile_options(renderer_policy_test PRIVATE -Wall -Wextra -Werror)
target_include_directories(renderer_policy_test PRIVATE ..)
target_link_libraries(renderer_policy_test PRIVATE PkgConfig::GLIB)
add_test(NAME renderer_policy COMMAND renderer_policy_test)

add_executable(window_options_test window_options_test.cc ../window_options.cc)
target_compile_features(window_options_test PRIVATE cxx_std_14)
target_compile_options(window_options_test PRIVATE -Wall -Wextra -Werror)
target_include_directories(window_options_test PRIVATE ..)
target_link_libraries(window_options_test PRIVATE PkgConfig::GTK)
add_test(NAME window_options COMMAND window_options_test)
set_tests_properties(window_options PROPERTIES SKIP_RETURN_CODE 77)
